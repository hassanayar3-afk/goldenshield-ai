=== ENSURE_BOT ===
$ErrorActionPreference="SilentlyContinue"
$officialPy = "C:\saqr\.venv\Scripts\python.exe"
# kill ALL uvicorn app.main first
Get-CimInstance Win32_Process -ErrorAction SilentlyContinue | Where-Object {
  $_.CommandLine -and ($_.CommandLine -like "*uvicorn*") -and ($_.CommandLine -like "*app.main*")
} | ForEach-Object {
  if ($_.CommandLine -notlike "*C:\saqr\.venv\Scripts\python.exe*") {
    Stop-Process -Id $_.ProcessId -Force -ErrorAction SilentlyContinue
  }
}
Start-Sleep -Milliseconds 500
$aliveOfficial = $false
Get-CimInstance Win32_Process -ErrorAction SilentlyContinue | Where-Object {
  $_.CommandLine -and ($_.CommandLine -like "*C:\saqr\.venv\Scripts\python.exe*") -and ($_.CommandLine -like "*uvicorn*")
} | ForEach-Object { $aliveOfficial = $true }
if (-not $aliveOfficial) {
  Get-NetTCPConnection -LocalPort 8090 -ErrorAction SilentlyContinue | ForEach-Object {
    Stop-Process -Id $_.OwningProcess -Force -ErrorAction SilentlyContinue
  }
  Start-Sleep 1
  if (Test-Path $officialPy) {
    New-Item -ItemType Directory -Force -Path "C:\saqr\storage" | Out-Null
    $env:PORT="8090"
    Start-Process -FilePath $officialPy -ArgumentList "-m","uvicorn","app.main:app","--host","0.0.0.0","--port","8090" -WorkingDirectory "C:\saqr" -WindowStyle Hidden -RedirectStandardOutput "C:\saqr\storage\bot_out.log" -RedirectStandardError "C:\saqr\storage\bot_err.log"
  }
}
